const { Client } = require('pg');

(async () => {

  const newPassword = 'NewPass123!';
  const hash = await hashPassword(newPassword);
  console.log('generated hash:', hash);

  const client = new Client({ connectionString: process.env.DATABASE_URL });
  await client.connect();

  const userRes = await client.query('SELECT id FROM users WHERE email = $1', [email]);
  if (userRes.rows.length !== 1) {
    console.log('ABORT: expected 1 user, found', userRes.rows.length);
    await client.end();
    return;
  }
  const userId = userRes.rows[0].id;

  await client.query(
    "UPDATE accounts SET password = $1 WHERE user_id = $2 AND provider_id = 'credential'",
    [hash, userId]
  );

  const check = await client.query(
    "SELECT password FROM accounts WHERE user_id = $1 AND provider_id = 'credential'",
    [userId]
  );
  const ok = await verifyPassword({ hash: check.rows[0].password, password: newPassword });
  console.log('round-trip verify result:', ok);

  await client.end();
})();
EOF
